{{- /* A screenshot's sizes: (screenshot number) → dict srcset, fallback (1080w), full */ -}} {{- $img := resources.Get (printf "images/screenshots/%v.png" .) -}} {{- if not $img }}{{ errorf "screenshot: no assets/images/screenshots/%v.png" . }}{{ end -}} {{- $srcset := slice -}} {{- $fallback := "" -}} {{- range slice 400 700 1080 1400 1792 -}} {{- if le . $img.Width -}} {{- $r := $img.Resize (printf "%dx webp q82" .) -}} {{- $srcset = $srcset | append (printf "%s %dw" $r.RelPermalink .) -}} {{- if eq . 1080 }}{{ $fallback = $r }}{{ end -}} {{- end -}} {{- end -}} {{- $full := $img.Resize (printf "%dx webp q88" $img.Width) -}} {{- return dict "srcset" (delimit $srcset ", ") "fallback" $fallback "full" $full -}}
